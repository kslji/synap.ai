# Deploy the Surf AI marketing site, then keep the auth host running.
# The portable zip pack is retired. This script does not build or publish it.
# /var/www/synap.surf receives the static site (macOS .dmg and Windows .exe only).
#
# The new desktop app and this website do not call the auth host.
# apps/host stays up for existing OTP / feedback accounts until that API is replaced.
#
# Required in ~/surf.ai/.env if the host is still used:
#   HOST_BIND=127.0.0.1:18765
#   PLATFORM_DATA_DIR=/var/lib/synap/platform
#   CORS_ORIGINS=https://synap.surf,https://www.synap.surf
#   ADMIN_EMAILS=...
#   ADMIN_PASSWORD='...'

cd ~/surf.ai && git pull origin main

# Keep systemd unit in sync (auth host)
sudo cp ~/surf.ai/apps/host/deploy/synap-host.service /etc/systemd/system/synap-host.service
sudo systemctl daemon-reload

cd ~/surf.ai && npm ci
npm run build -w @surf/web
rsync -a --delete apps/web/dist/ /var/www/synap.surf/ || (sudo rm -rf /var/www/synap.surf/* && sudo cp -a apps/web/dist/. /var/www/synap.surf/)

mkdir -p ~/surf.ai/apps/host/logs
sudo systemctl restart synap-host
sleep 2
sudo systemctl is-active synap-host
curl -sS http://127.0.0.1:18765/health | head -c 200; echo
curl -sS -o /dev/null -w "site %{http_code}\n" https://synap.surf/

# If the host health check fails:
#   tail -n 80 ~/surf.ai/apps/host/logs/error.log
