# Deploy static shell + restart thin auth host (Local-First).
# AI stays on each user's machine. VPS = accounts/OTP/feedback only.
#
# Required in ~/surf.ai/.env :
#   HOST_BIND=127.0.0.1:18765
#   PLATFORM_DATA_DIR=/var/lib/synap/platform
#   CORS_ORIGINS=https://synap.surf,https://www.synap.surf
#   ADMIN_EMAILS=...
#   ADMIN_PASSWORD='...'

cd ~/surf.ai && git pull origin main

# Moss pack vault: seal script reads ~/surf.ai/.env (MOSS_PROJECT_ID / MOSS_PROJECT_KEY).
# Confirm on the VPS (shows only whether set, not the secret):
#   grep -E '^MOSS_PROJECT_(ID|KEY)=' ~/surf.ai/.env | sed 's/=.*/=***set***/'
# Or:  cd ~/surf.ai/apps/web && npm run seal:moss
# Expect: "sealed Moss vault → ... (credentials encrypted; not plaintext)"

# Keep systemd unit in sync (auth host)
sudo cp ~/surf.ai/apps/host/deploy/synap-host.service /etc/systemd/system/synap-host.service
sudo systemctl daemon-reload

cd ~/surf.ai/apps/web && npm ci
npm run build
rsync -a --delete out/ /var/www/synap.surf/ || (sudo rm -rf /var/www/synap.surf/* && sudo cp -a out/. /var/www/synap.surf/)

mkdir -p ~/surf.ai/apps/host/logs
sudo systemctl restart synap-host
sleep 2
sudo systemctl is-active synap-host
curl -sS http://127.0.0.1:18765/health | head -c 200; echo
curl -sS https://synap.surf/health | head -c 200; echo

# If health fails:
#   tail -n 80 ~/surf.ai/apps/host/logs/error.log
#   cd ~/surf.ai/apps/host && .venv/bin/python -c "import main; print('ok')"
