# Deploy the Surf AI marketing site and the services/api stack.
# The portable zip pack is retired. This script does not build or publish it.
# /var/www/synap.surf receives the static site (macOS .dmg and Windows .exe only).
# The legacy auth host (apps/host) is gone; the API runs from deploy/docker-compose.yml.

cd ~/surf.ai && git pull origin main

cd ~/surf.ai && npm ci
npm run build -w @surf/web
rsync -a --delete apps/web/dist/ /var/www/synap.surf/ || (sudo rm -rf /var/www/synap.surf/* && sudo cp -a apps/web/dist/. /var/www/synap.surf/)

# Backend stack (Caddy, API, Postgres/pgvector, Valkey). deploy/.env holds real values.
cd ~/surf.ai/deploy && docker compose up -d --build

curl -sS -o /dev/null -w "site %{http_code}\n" https://synap.surf/
curl -sS https://api.synap.surf/v1/health | head -c 200; echo

# If the API health check fails:
#   cd ~/surf.ai/deploy && docker compose logs --tail 80 api
